function FindProxyForURL(url, host) {

  if (dnsDomainIs(host, ".[DOMAIN_1]") ||
      dnsDomainIs(host, "*.[DOMAIN_2]") ||
      dnsDomainIs(host, ".[DOMAIN_3]") ||
