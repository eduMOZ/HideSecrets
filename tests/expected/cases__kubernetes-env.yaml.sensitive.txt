# Environment of a deployment: name / value pairs (kubectl get deploy -o yaml, a pod spec, Helm extraEnv).
# A value is masked when the NAME of the variable says it is a secret. The other variables stay readable.
apiVersion: apps/v1
kind: Deployment
metadata:
  name: nimbus-hub
spec:
  template:
    spec:
      containers:
        - name: hub
          image: registry.nimbus-cloud.test/nimbus/hub:4.2.1
          env:
            - name: NIMBUS_DOMAIN
              value: [DOMAIN_1]
            - name: NIMBUS_OIDC_CLIENT_SECRET
              value: [SECRET_1]
            - name: NIMBUS_COOKIE_KEY
              value: "[SECRET_2]"
            - name: NIMBUS_DB_PASSWORD
              value: '[SECRET_3]'
            - name: NIMBUS_S3_SECRET_ACCESS_KEY
              value: [SECRET_4]   # rotated yesterday
            - name: NIMBUS_PUSH_AUTH_KEY
              valueFrom:
                secretKeyRef:
                  name: nimbus-secrets
                  key: push-auth-key
            - name: NIMBUS_REDIS_PASSWORD
              value: ${REDIS_PASSWORD}
            - name: LOG_LEVEL
              value: debug
            - name: NIMBUS_REPLICAS
              value: "3"
# Helm values
extraEnv:
  - name: NIMBUS_SYNC_MASTER_SECRET
    value: [SECRET_5]
  - { name: NIMBUS_MEGAPHONE_TOKEN, value: [SECRET_6] }
  - {"name": "NIMBUS_BROWSER_BASE_KEY", "value": "[SECRET_7]"}
